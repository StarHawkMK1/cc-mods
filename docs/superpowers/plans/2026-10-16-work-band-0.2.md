# work-band 0.2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** work-band 0.1.0에 다른 스킬 표시, `--resume` 흐름 복원, 테스트 빨강/초록 칩, 요청 경과 시간을 더해 0.2.0으로 만든다.

**Architecture:** 상태 변화 규칙을 순수 함수 모듈 `events.ts`(`Work → Work`)로 모으고, 실시간 훅(`work-band.tsx`)과 resume 재생(`replay.ts`)이 같은 함수를 접는다. 테스트 판정은 `tests.ts`, 막대 계산은 `flow.ts`, 그리기는 `draw.ts`에 둔다. 훅은 0.1처럼 이미 일어나는 이벤트를 읽기만 하고, 새로 `tool.call {Bash}`, `skill.prompt`, `classic.SessionStart`를 읽는다.

**Tech Stack:** Claude Code mod 훅 API(`claude-code`), TSX/TS 모듈, 테스트 키트 `claude-code/testing` (`claude plugin test`)

**Spec:** `docs/superpowers/specs/2026-10-16-work-band-0.2-design.md` (0.1 규칙은 `docs/superpowers/specs/2026-10-15-work-band-design.md`)

```tsx
import type { EngineInterface, Register } from 'claude-code'

// Appends one JSON line per event to $WB_PROBE_LOG.
async function log($: EngineInterface, what: string, data: unknown): Promise<void> {
  const path = await $.env.get('WB_PROBE_LOG')
  if (!path) return
  let prev = ''
  try {
    prev = String(await $.fs.read(path))
  } catch {
    // First line.
  }
  await $.fs.write(path, prev + JSON.stringify({ what, data }) + '\n')
}

const head = (v: unknown): unknown => (typeof v === 'string' ? v.slice(0, 200) : v)

// The last rows of the main transcript, cut short.
async function transcript($: EngineInterface): Promise<unknown> {
  try {
    const rows = await $.session.messages()
    return {
      count: rows.length,
      rows: rows.slice(-14).map(r => ({
        role: r.role,
        text: head(r.text),
        tools: r.toolUses.map(u => ({ tool: u.tool, isError: u.isError, result: head(typeof u.result === 'string' ? u.result : JSON.stringify(u.result)) })),
        results: r.toolResults?.length ?? 0,
      })),
    }
  } catch (err) {
    return `messages() failed: ${String(err)}`
  }
}

export const register: Register = on => {
  on('session.start', async ($, e, next) => {
    const r = await next(e)
    await log($, 'session.start', { isInteractive: e.isInteractive, transcript: await transcript($) })
    return r
  })
  on('classic.SessionStart', async ($, e, next) => {
    const r = await next(e)
    await log($, 'classic.SessionStart', { source: e.source, transcript: await transcript($) })
    return r
  })
  on('prompt.submit', async ($, e, next) => {
    await log($, 'prompt.submit', { command: e.command, turnId: e.turnId, origin: e.origin?.kind, text: head(e.text) })
    return next(e)
  })
  on('command.run', async ($, e, next) => {
    await log($, 'command.run', { command: e.command, args: e.args, origin: e.origin?.kind })
    return next(e)
  })
  on('skill.prompt', async ($, e, next) => {
    await log($, 'skill.prompt', { skill: e.skill, pluginName: e.pluginName, source: e.source })
    return next(e)
  })
  on('tool.call', { tool: 'Bash' }, async ($, e, next) => {
    const r = await next(e)
    const res = r.result as unknown
    await log($, 'bash', {
      command: (e as unknown as { command?: unknown }).command,
      isError: r.isError,
      resultType: typeof res,
      keys: res && typeof res === 'object' ? Object.keys(res) : null,
      result: head(typeof res === 'string' ? res : JSON.stringify(res)),
      text: head(r.text),
    })
    return r
  })
  on('turn.complete', async ($, e, next) => {
    if (!e.agentId) await log($, 'turn.complete', { reason: e.reason, transcript: await transcript($) })
    return next(e)
  })
}
```

`mods/work-band/hooks/events.ts`:

```ts
import type { Work } from '../types'
import { advance, cleanTodos, createTodo, listOf, onWrite, skillStage, startRequest, updateTodo, withList, type TaskUpdateInput } from './flow'

// How the work changes with each signal. The live hooks and the replay of a
// resumed session's transcript fold the same functions, so each rule lives once.

// A superpowers stage skill moves the flow; any other skill changes nothing.
export const onSkill = (w: Work, name: unknown, args: unknown): Work => {
  const stage = skillStage(name)
  return stage ? { ...w, flow: advance(w.flow, stage, args) } : w
}

// A file the main session wrote or edited (`content` is Write's).
export const onFile = (w: Work, path: unknown, content: unknown, root?: string): Work => {
  const flow = onWrite(w.flow, path, content, root)
  return flow === w.flow ? w : { ...w, flow }
}

export const onTodoWrite = (w: Work, agentId: string | undefined, todos: unknown): Work => withList(w, agentId, cleanTodos(todos))

export const onTaskCreate = (w: Work, agentId: string | undefined, id: unknown, subject: unknown, activeForm: unknown): Work => {
  const todos = createTodo(listOf(w, agentId), id, subject, activeForm)
  return todos ? withList(w, agentId, todos) : w
}

export const onTaskUpdate = (w: Work, agentId: string | undefined, input: TaskUpdateInput): Work => {
  const todos = updateTodo(listOf(w, agentId), input)
  return todos ? withList(w, agentId, todos) : w
}

  const { [agentId]: _gone, ...rest } = w.agentTodos
  return { ...w, agentTodos: rest }
```

```ts
// A prompt the user sent while idle: 0.1's rules, the request's clock starts,
// and a slash command still waiting is stale.
export const onRequest = (w: Work, at: number): Work => {
  const { pendingSlash: _stale, ...rest } = startRequest(w, at)
  return { ...rest, clock: { from: at } }
}

// The main turn ended at `at`: the request's clock stops there if it still ran.
// A slash command sent while idle that was never expanded is stale; one typed
// into the turn waits in the queue and expands after it, so it stays. The
// finishing skill ends its turn asking merge / PR / keep; the turn that
// answered the user's reply ends the flow.
export const onMainTurnEnd = (w: Work, at: number, isAnswer: boolean): Work => {
  const { pendingSlash, ...rest } = w
  const clock = rest.clock && rest.clock.to === undefined ? { ...rest.clock, to: at } : rest.clock
  const flow = isAnswer && rest.flow?.stage === 'finish' && rest.flow.hasReply && !rest.flow.isFinished ? { ...rest.flow, isFinished: true } : rest.flow
  return { ...rest, flow, ...(clock ? { clock } : {}), ...(pendingSlash && !pendingSlash.isIdle ? { pendingSlash } : {}) }
}
```

```ts
// ---------------------------------------------------------------------------
// Slash commands: a skill only once `skill.prompt` expands it.

const slashName = (v: unknown): string => (typeof v === 'string' ? v.trim().replace(/^\//, '') : '')
const keyOf = (v: unknown): string => slashName(v).toLowerCase()

// A slash command the user sent; `isIdle` when no turn was running. The same
// command again (command.run after prompt.submit) keeps the first.
export const onSlash = (w: Work, name: unknown, args: unknown, isIdle: boolean): Work => {
  const key = keyOf(name)
  if (!key || keyOf(w.pendingSlash?.name) === key) return w
  return { ...w, pendingSlash: { name: slashName(name), args: typeof args === 'string' ? args : '', isIdle } }
}

// The names a `skill.prompt` may go by: as given, and with its plugin in front.
const promptedKeys = (skill: unknown, pluginName: unknown): string[] => {
  const s = keyOf(skill)
  const p = keyOf(pluginName)
  return p && s && !s.includes(':') ? [s, `${p}:${s}`] : [s]
}

// `skill.prompt` for the waiting slash command: it was a skill. An idle one is a
// new request first, then the skill applies. Any other `skill.prompt` (the
// model's Skill call, a subagent's) changes nothing here.
export const confirmSlash = (w: Work, skill: unknown, pluginName: unknown, at: number): { work: Work; slash?: SlashCandidate } => {
  const c = w.pendingSlash
  if (!c || !promptedKeys(skill, pluginName).includes(keyOf(c.name))) return { work: w }
  const { pendingSlash: _done, ...rest } = w
  const base = c.isIdle ? onRequest(rest, at) : rest
  return { work: onSkill(base, c.name, c.args), slash: c }
}
```

```ts
// Is a Bash command a test run, and did it pass? From the exit code where it is
// the test's own, else from the summary the runner printed.

export type Verdict = 'pass' | 'fail'

export const TEST_COMMANDS: readonly string[] = [
  'npm test', 'npm run test', 'npm t', 'pnpm test', 'pnpm run test', 'yarn test', 'bun test', 'bun run test',
  'vitest', 'jest', 'mocha', 'playwright test', 'deno test',
  'pytest', 'python -m pytest', 'python3 -m pytest', 'py.test', 'tox',
  'cargo test', 'cargo nextest', 'go test', 'dotnet test', 'swift test', 'ctest', 'make test',
  'mvn test', 'gradle test', './gradlew test', 'rspec', 'rake test', 'phpunit', 'mix test',
  'claude plugin test',
]

// A comma-separated setting: its trimmed, non-empty entries.
export const parseList = (v: unknown): string[] =>
  typeof v === 'string' ? v.split(',').map(s => s.trim()).filter(Boolean) : []

export type Piece = { text: string; next: string }

// The command cut at &&, ||, ; and |; `next` is the operator after each piece ('' for the last).
export const piecesOf = (command: string): Piece[] => {
  const parts = command.split(/(&&|\|\||;|\|)/)
  const out: Piece[] = []
  for (let i = 0; i < parts.length; i += 2) out.push({ text: parts[i] ?? '', next: parts[i + 1] ?? '' })
  return out
}

const ENV = /^[A-Za-z_][A-Za-z0-9_]*=\S*\s+/
const RUNNER = /^(?:npx|bunx|pnpm\s+(?:exec|dlx)|uv\s+run|poetry\s+run|bundle\s+exec)\s+/

// A piece without what runs before the command itself: `cd dir`, `VAR=value`, a runner.
export const bareOf = (piece: string): string => {
  let s = piece.trim().replace(/\s+/g, ' ')
  if (/^cd(\s|$)/.test(s)) return ''
  for (let prev = ''; prev !== s; ) {
    prev = s
    s = s.replace(ENV, '').replace(RUNNER, '')
  }
  return s
}

// The piece starts with a test command, word for word.
export const isTestRun = (piece: string, extra: readonly string[] = []): boolean => {
  const s = bareOf(piece).toLowerCase()
  if (!s) return false
  return [...TEST_COMMANDS, ...extra].some(p => {
    const q = p.trim().toLowerCase()
    return !!q && (s === q || s.startsWith(`${q} `))
  })
}

// Failure first: a failed count, a FAIL at a line's head (go, jest), "Tests failed".
const FAILED = [/\b[1-9]\d*\s+(?:failed|failing|failures?|fail)\b/i, /^\s*(?:FAIL|FAILED)\b/m, /^--- FAIL:/m, /\btests failed\b/i]
const PASSED = [/\b\d+\s+(?:passed|pass)\b/i, /^ok\s/m, /\btests passed\b/i, /\b0\s+failed\b/i]

export const verdictOf = (output: string): Verdict | null =>
  FAILED.some(r => r.test(output)) ? 'fail' : PASSED.some(r => r.test(output)) ? 'pass' : null

export type BashCall = { command?: unknown; run_in_background?: unknown }
export type BashAnswer = { deny?: string; isError?: boolean; result?: unknown; text?: string }

// What the command printed: an errored call's text, or the record's stdout and stderr, and what the model read.
const outputOf = (r: BashAnswer): string => {
  const res = r.result
  const parts: unknown[] =
    typeof res === 'string' ? [res] : res && typeof res === 'object' ? [(res as { stdout?: unknown }).stdout, (res as { stderr?: unknown }).stderr] : []
  return [...parts, r.text].filter((p): p is string => typeof p === 'string').join('\n')
}

// 'pass' or 'fail' for a test run that finished in the foreground; null for anything else.
export const testVerdict = (call: BashCall, r: BashAnswer, extra: readonly string[] = []): Verdict | null => {
  if (r.deny !== undefined || call.run_in_background === true || typeof call.command !== 'string') return null
  const res = (r.result && typeof r.result === 'object' ? r.result : {}) as { interrupted?: unknown; backgroundTaskId?: unknown; timedOutAfterMs?: unknown }
  if (res.interrupted === true || res.backgroundTaskId || res.timedOutAfterMs) return null
  const pieces = piecesOf(call.command)
  let at = -1
  pieces.forEach((p, i) => {
    if (isTestRun(p.text, extra)) at = i
  })
  if (at < 0) return null
  // `a && b` stops at a failure, so the exit code is still the test's; `|`, `;` and `||` hide it.
  const isOwnExit = pieces.slice(at).every(p => p.next === '' || p.next === '&&')
  return isOwnExit ? (r.isError ? 'fail' : 'pass') : verdictOf(outputOf(r))
}
```

```ts
  const parts = name.split(':')

export const isShownSkill = (name: string, skills: readonly string[]): boolean => {
  const n = name.toLowerCase()
  return !skills.length || skills.some(raw => {
    const k = raw.toLowerCase()
    return k.endsWith(':') ? n.startsWith(k) : n === k
  })

// else Running until the main turn ends.
export const skillBar = (w: Work, runs: AgentRun[], s: Strings): Bar | null => {

  const note = w.skill.args ? { note: w.skill.args } : {}
  const base = todoBar(w.todos, s) ?? agentsBar(requestAgents(runs, w.since), s)
  if (base) return { ...base, title, ...note }
```

```ts
export const onSkill = (w: Work, name: unknown, args: unknown, skills: readonly string[]): Work => {
```

```ts
import type { Work } from '../types'
import { onFile, onMainTurnEnd, onRequest, onSkill, onTaskCreate, onTaskUpdate, onTest, onTodoWrite } from './events'
import { blankWork, mdPathOf, skillStage, type TaskUpdateInput } from './flow'
import { testVerdict } from './tests'

// A resumed or branched session starts with empty state: its transcript, read
// back through the same rules the live hooks use, rebuilds what outlives a request.

type Use = { tool?: unknown; input?: unknown; result?: unknown; text?: unknown; isError?: unknown }
type Message = { role?: unknown; text?: unknown; toolUses?: unknown; toolResults?: unknown }

export type Signal =
  | { kind: 'request' }
  | { kind: 'slash'; name: string; args: string }
  | { kind: 'tool'; tool: string; input: Record<string, unknown>; result?: unknown; text?: string; isError: boolean }
  | { kind: 'answer' }

// User rows that are not a prompt the user wrote.
const NOT_PROMPT = ['<task-notification', '<system-reminder', '<local-command', 'Caveat:', 'Base directory for this skill:', '[Request interrupted']
const COMMAND = /<command-name>\/?([^<\s]+)\s*<\/command-name>/
const COMMAND_ARGS = /<command-args>([\s\S]*?)<\/command-args>/

export const signalsOf = (messages: unknown): Signal[] => {
  const out: Signal[] = []
  for (const m of Array.isArray(messages) ? (messages as unknown[]) : []) {
    if (!m || typeof m !== 'object') continue
    const { role, text: rawText, toolUses, toolResults } = m as Message
    const text = typeof rawText === 'string' ? rawText.trim() : ''
    if (role === 'user') {
      if (Array.isArray(toolResults) && toolResults.length) continue
      const command = COMMAND.exec(text)?.[1]
      if (command) {
        // Only a superpowers stage skill is known to be a skill here: any other
        // slash command may be a built-in (/model), and other skills are not restored.
        if (skillStage(command)) out.push({ kind: 'request' }, { kind: 'slash', name: command, args: COMMAND_ARGS.exec(text)?.[1]?.trim() ?? '' })
        continue
      }
      if (text && !NOT_PROMPT.some(p => text.startsWith(p))) out.push({ kind: 'request' })
    } else if (role === 'assistant') {
      const uses = Array.isArray(toolUses) ? (toolUses as Use[]) : []
      for (const u of uses) {
        if (!u || typeof u.tool !== 'string') continue
        out.push({
          kind: 'tool',
          tool: u.tool,
          input: u.input && typeof u.input === 'object' ? (u.input as Record<string, unknown>) : {},
          result: u.result,
          ...(typeof u.text === 'string' ? { text: u.text } : {}),
          isError: u.isError === true,
        })
      }
      // A reply with no tool calls ends the turn with an answer.
      if (!uses.length && text) out.push({ kind: 'answer' })
    }
  }
  return out
}

export type ReplayOptions = { root?: string; skills: readonly string[]; tests: readonly string[]; at: number }

export const replay = (messages: unknown, o: ReplayOptions): { work: Work; planPath?: string } => {
  let w = blankWork()
  let planPath: string | undefined
  const skillSignal = (name: unknown, args: unknown) => {
    w = onSkill(w, name, args, o.skills)
    if (skillStage(name) === 'build') planPath = mdPathOf(args) ?? planPath
  }
  for (const s of signalsOf(messages)) {
    if (s.kind === 'request') w = onRequest(w, 0)
    else if (s.kind === 'slash') skillSignal(s.name, s.args)
    else if (s.kind === 'answer') w = onMainTurnEnd(w, 0, true)
    else if (s.tool === 'Bash') {
      const verdict = testVerdict(s.input, { isError: s.isError, result: s.result, ...(s.text !== undefined ? { text: s.text } : {}) }, o.tests)
      if (verdict) w = onTest(w, verdict === 'pass', 0)
    } else if (s.isError || (s.result as { success?: unknown } | undefined)?.success === false) continue
    else if (s.tool === 'Skill') skillSignal(s.input.skill, s.input.args)
    else if (s.tool === 'Write') w = onFile(w, s.input.file_path, s.input.content, o.root)
    else if (s.tool === 'Edit') w = onFile(w, s.input.file_path, undefined, o.root)
    else if (s.tool === 'NotebookEdit') w = onFile(w, s.input.notebook_path, undefined, o.root)
    else if (s.tool === 'TodoWrite') w = onTodoWrite(w, undefined, s.input.todos)
    else if (s.tool === 'TaskCreate') w = onTaskCreate(w, undefined, (s.result as { task?: { id?: unknown } } | undefined)?.task?.id, s.input.subject, s.input.activeForm)
    else if (s.tool === 'TaskUpdate') w = onTaskUpdate(w, undefined, s.input as TaskUpdateInput)
  }
  // Only what outlives a request: the flow, an unfinished list, and the test result while the flow is under way.
  const flow = w.flow
  const isLive = !!flow && !flow.isFinished
  const work: Work = {
    ...blankWork(),
    flow,
    todos: w.todos.every(t => t.status === 'completed') ? [] : w.todos,
    since: o.at,
    isReplayed: true,
    ...(isLive && w.test ? { test: w.test } : {}),
  }
  return isLive && flow?.stage === 'build' && !flow.planTasks && planPath ? { work, planPath } : { work }
}
```

```tsx
// A resumed or branched session starts with empty state: rebuild what outlives
// a request from the transcript, once. Live work already there stands, and so
// does anything written while the transcript was read.
async function restore($: EngineInterface, skills: readonly string[], tests: readonly string[]): Promise<void> {
  const held = await $.state.get({ plugin: 'work-band', key: 'work' })
  const w0 = held.value ?? blankWork()
  if (w0.isReplayed) return
  const marked = await $.state.set({ plugin: 'work-band', key: 'work' }, { ...w0, isReplayed: true }, { ifVersion: held.version })
  if (!marked.isSet || w0.flow || w0.todos.length) return
  try {
    const messages = await $.session.messages()
    const root = await $.session.root().catch(() => undefined)
    const at = await $.clock.now()
    const { work: restored, planPath } = replay(messages, { root, skills, tests, at })
    if (!restored.flow && !restored.todos.length) return
    const wrote = await $.state.set({ plugin: 'work-band', key: 'work' }, restored, { ifVersion: marked.version })
    if (wrote.isSet && planPath) await readPlan($, planPath)
  } catch {
    // No transcript to read: the band waits for live events.
  }
}
```
